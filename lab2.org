#+begin_src python :results output :hlines
import random
import time
import numpy as np
import matplotlib
matplotlib.use('TkAgg')
import matplotlib.pyplot as plt

def kmeans(points: np.ndarray, centroids: np.ndarray, EPS: float) -> tuple[np.ndarray, np.ndarray, int, list[tuple[np.ndarray, np.ndarray]]]:
    snapshots = []
    iters = 0
    while True:
        iters += 1
        dists = np.linalg.norm(points[:, None, :] - centroids[None, :, :], axis=2)
        labels = np.argmin(dists, axis=1)
        snapshots.append((labels.copy(), centroids.copy()))
        means = np.array([points[labels == i].mean(axis=0)
                          if np.any(labels == i) else centroids[i]
                          for i in range(k)])
        if np.allclose(means, centroids, atol=EPS):
            centroids = means
            break
        centroids = means
    return labels, centroids, iters, snapshots


seed = 52
random.seed(seed)
np.random.seed(seed)


k = random.randint(2, 5)
points = np.random.uniform(-10, 10, size=(100, 2))
centroids = np.random.uniform(-10, 10, size=(k, 2))

labels, final_centroids, iters, snapshots = kmeans(points, centroids, EPS=1e-6)

fig, axes = plt.subplots(2, 2, figsize=(10, 10))
axes = axes.flatten()
cmap = matplotlib.colormaps.get_cmap('tab10').resampled(k)

axes[0].set_title('Points')
axes[0].scatter(points[:, 0], points[:, 1], s=30)

for i, (s_labels, s_centroids) in enumerate(snapshots[:2], start=1):
    axes[i].set_title(f'Iteration {i}')
    axes[i].scatter(points[:, 0], points[:, 1], c=s_labels, cmap=cmap, s=30)
    axes[i].scatter(s_centroids[:, 0], s_centroids[:, 1], marker='x', s=120, color='black', linewidths=2)

last_labels, last_centroids = snapshots[-1]
axes[3].set_title(f'Iteration {iters} (last)')
axes[3].scatter(points[:, 0], points[:, 1], c=last_labels, cmap=cmap, s=30)
axes[3].scatter(last_centroids[:, 0], last_centroids[:, 1], marker='x', s=120, color='black', linewidths=2)

plt.tight_layout()
print(f"iterations {iters}")
print("centroids")
print(final_centroids)

plt.show()
